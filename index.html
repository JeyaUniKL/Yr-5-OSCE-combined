// BULLETPROOF STATION DETECTOR (Scans search, hash, and full URL) function detectStationFromURL() { const fullUrl = window.location.href.toLowerCase(); // Check hash first (e.g., #dengue or #station=dengue) let hash = window.location.hash.toLowerCase().replace(/^[#\/]+/, ''); if (hash.startsWith('station=')) hash = hash.replace('station=', ''); // Check query param (e.g., ?station=dengue or ?dengue) const urlParams = new URLSearchParams(window.location.search); let query = (urlParams.get('station') || '').toLowerCase().trim(); // Mapping table const targets = [ { key: "dengue_fever", terms: ["dengue"] }, { key: "febrile_convulsion", terms: ["febrile", "seizure", "convulsion"] }, { key: "asthma_discharge", terms: ["asthma"] }, { key: "neonatal_jaundice", terms: ["nnj", "jaundice"] }, { key: "hfmd", terms: ["hfmd"] }, { key: "bronchiolitis", terms: ["bronch"] }, { key: "gastroenteritis", terms: ["gastro", "age", "diarrhea"] }, { key: "thalassemia", terms: ["thal"] }, { key: "g6pd", terms: ["g6pd"] }, { key: "nephrotic", terms: ["nephrotic"] } ]; // 1. Check hash and query parameter first const paramToTest = hash || query; if (paramToTest) { for (const t of targets) { if (t.terms.some(term => paramToTest.includes(term))) { return t.key; } } } // 2. Scan entire URL string for safety for (const t of targets) { if (t.terms.some(term => fullUrl.includes(term))) { return t.key; } } return null; // Practice mode } window.onload = () => { if (!window.OSCE_STATIONS || Object.keys(window.OSCE_STATIONS).length === 0) { alert("Error: stations.js failed to load. Ensure stations.js is in the same directory."); return; } const keyInput = document.getElementById('api-key-input'); if (apiKey) keyInput.value = apiKey; const idInput = document.getElementById('candidate-id'); if (storedCandidateId) idInput.value = storedCandidateId; const select = document.getElementById('station-select'); select.innerHTML = ''; for (const key in window.OSCE_STATIONS) { const opt = document.createElement('option'); opt.value = key; opt.textContent = window.OSCE_STATIONS[key].title; select.appendChild(opt); } // Run bulletproof detector const matchedKey = detectStationFromURL(); if (matchedKey && window.OSCE_STATIONS[matchedKey]) { // --- EXAM LOCK MODE ENGAGED --- isExamMode = true; select.style.display = "none"; // Hide dropdown completely const banner = document.getElementById('exam-lock-banner'); banner.style.display = "inline-block"; document.getElementById('locked-station-name').innerText = window.OSCE_STATIONS[matchedKey].title; loadStation(matchedKey); } else { // --- PRACTICE MODE (Dropdown active) --- isExamMode = false; select.style.display = "inline-block"; document.getElementById('exam-lock-banner').style.display = "none"; const defaultKey = Object.keys(window.OSCE_STATIONS)[0]; select.value = defaultKey; loadStation(defaultKey); } setupSpeechRecognition(); };
